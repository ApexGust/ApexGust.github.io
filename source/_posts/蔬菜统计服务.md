<script>
    (function() {
        const uploadArea = document.getElementById('uploadArea');
        const fileInput = document.getElementById('fileInput');
        const statsContainer = document.getElementById('statsContainer');
        const statsBody = document.getElementById('statsBody');
        const errorMessage = document.getElementById('errorMessage');
        const successMessage = document.getElementById('successMessage');
        const fileInfo = document.getElementById('fileInfo');
        const loading = document.getElementById('loading');
        const totalInfo = document.getElementById('totalInfo');
        const copyBtn = document.getElementById('copyBtn');
        const copyIcon = document.getElementById('copyIcon');
        const copyText = document.getElementById('copyText');

        let currentStats = null;
        let currentUnit = '';
        let currentFileName = '';
        let priceData = {}; // 存储每个品名的单价

        uploadArea.addEventListener('click', () => {
            fileInput.click();
        });

        fileInput.addEventListener('change', (e) => {
            const file = e.target.files[0];
            if (file) {
                handleFile(file);
            }
        });

        uploadArea.addEventListener('dragover', (e) => {
            e.preventDefault();
            uploadArea.classList.add('dragover');
        });

        uploadArea.addEventListener('dragleave', () => {
            uploadArea.classList.remove('dragover');
        });

        uploadArea.addEventListener('drop', (e) => {
            e.preventDefault();
            uploadArea.classList.remove('dragover');
            const file = e.dataTransfer.files[0];
            if (file) {
                handleFile(file);
            }
        });

        copyBtn.addEventListener('click', () => {
            if (!currentStats || currentStats.length === 0) {
                showError('没有可复制的数据');
                return;
            }
            copyResultsToClipboard();
        });

        function showError(message) {
            errorMessage.textContent = message;
            errorMessage.style.display = 'block';
            successMessage.style.display = 'none';
            setTimeout(() => {
                errorMessage.style.display = 'none';
            }, 5000);
        }

        function showSuccess(message) {
            successMessage.textContent = message;
            successMessage.style.display = 'block';
            errorMessage.style.display = 'none';
            setTimeout(() => {
                successMessage.style.display = 'none';
            }, 3000);
        }

        function showFileInfo(fileName, rowCount) {
            fileInfo.innerHTML = `📄 <strong>${fileName}</strong> - 共 ${rowCount} 条数据`;
            fileInfo.style.display = 'block';
        }

        function handleFile(file) {
            if (!file.name.match(/\.(xlsx|xls)$/i)) {
                showError('请上传 Excel 文件（.xlsx 或 .xls 格式）');
                return;
            }

            loading.style.display = 'block';
            statsContainer.style.display = 'none';

            const reader = new FileReader();
            reader.onload = (e) => {
                try {
                    const data = new Uint8Array(e.target.result);
                    const workbook = XLSX.read(data, { type: 'array' });
                    const firstSheetName = workbook.SheetNames[0];
                    const worksheet = workbook.Sheets[firstSheetName];
                    const jsonData = XLSX.utils.sheet_to_json(worksheet, { header: 1 });
                    
                    if (jsonData.length < 2) {
                        throw new Error('Excel文件中没有数据');
                    }

                    processData(jsonData, file.name);
                } catch (error) {
                    loading.style.display = 'none';
                    showError('解析Excel文件失败: ' + error.message);
                    console.error(error);
                }
            };

            reader.onerror = () => {
                loading.style.display = 'none';
                showError('读取文件失败');
            };

            reader.readAsArrayBuffer(file);
        }

        function processData(jsonData, fileName) {
            try {
                const headers = jsonData[0];
                const pinmingIndex = headers.findIndex(h => h && h.toString().includes('品名'));
                const quantityIndex = headers.findIndex(h => h && (h.toString().includes('购买量') || h.toString().includes('重量')));
                const unitIndex = headers.findIndex(h => h && h.toString().includes('计量单位'));
                const priceIndex = headers.findIndex(h => h && h.toString().includes('含税单价'));

                if (pinmingIndex === -1) {
                    throw new Error('未找到"品名"列，请检查Excel文件格式');
                }

                if (quantityIndex === -1) {
                    throw new Error('未找到"购买量"列，请检查Excel文件格式');
                }

                const stats = {};
                const prices = {}; // 存储每个品名的单价
                let totalRows = 0;
                let unit = '';

                for (let i = 1; i < jsonData.length; i++) {
                    const row = jsonData[i];
                    if (!row || row.length === 0) continue;

                    const pinming = row[pinmingIndex];
                    const quantity = row[quantityIndex];
                    
                    if (!pinming || quantity === undefined || quantity === null || quantity === '') {
                        continue;
                    }

                    if (unitIndex !== -1 && !unit && row[unitIndex]) {
                        unit = row[unitIndex].toString();
                    }

                    // 读取含税单价（优先从Excel获取）
                    if (priceIndex !== -1 && row[priceIndex] && !prices[pinming]) {
                        const price = parseFloat(row[priceIndex]);
                        if (!isNaN(price) && price > 0) {
                            prices[pinming] = price;
                        }
                    }

                    let numQuantity = parseFloat(quantity);
                    if (isNaN(numQuantity)) {
                        console.warn(`第 ${i + 1} 行的购买量无法转换为数字: ${quantity}`);
                        continue;
                    }

                    totalRows++;

                    if (!stats[pinming]) {
                        stats[pinming] = {
                            totalWeight: 0,
                            count: 0
                        };
                    }
                    stats[pinming].totalWeight += numQuantity;
                    stats[pinming].count++;
                }

                if (totalRows === 0) {
                    throw new Error('没有有效的数据行');
                }

                displayStats(stats, unit, totalRows, fileName, prices);
                showSuccess(`✅ 成功处理 ${totalRows} 条记录！`);
                showFileInfo(fileName, totalRows);

            } catch (error) {
                showError('处理数据失败: ' + error.message);
                console.error(error);
            } finally {
                loading.style.display = 'none';
            }
        }

        function displayStats(stats, unit, totalRows, fileName, excelPrices = {}) {
            const statsArray = Object.entries(stats).map(([name, data]) => ({
                name,
                totalWeight: data.totalWeight,
                count: data.count
            })).sort((a, b) => b.totalWeight - a.totalWeight);

            currentStats = statsArray;
            currentUnit = unit;
            currentFileName = fileName;
            priceData = {}; // 重置价格数据

            statsBody.innerHTML = '';

            const grandTotal = statsArray.reduce((sum, item) => sum + item.totalWeight, 0);

            statsArray.forEach((item, index) => {
                const excelPrice = excelPrices[item.name];
                const priceValue = excelPrice ? excelPrice.toFixed(2) : '';
                
                // 如果Excel中有单价，自动设置到priceData
                if (excelPrice) {
                    priceData[item.name] = excelPrice;
                }

                const row = document.createElement('tr');
                row.innerHTML = `
                    <td>${index + 1}</td>
                    <td><strong>${item.name}</strong></td>
                    <td class="weight-cell">${item.totalWeight.toFixed(2)} ${unit}</td>
                    <td>${item.count} 条</td>
                    <td>
                        <input type="number" 
                               class="price-input" 
                               placeholder="输入单价" 
                               step="0.01" 
                               min="0"
                               value="${priceValue}"
                               data-name="${item.name}"
                               data-weight="${item.totalWeight}">
                    </td>
                    <td class="amount-cell" data-name="${item.name}">${excelPrice ? '¥' + (item.totalWeight * excelPrice).toFixed(2) : '--'}</td>
                `;
                statsBody.appendChild(row);
            });

            // 更新总信息
            updateTotalInfo(grandTotal);

            // 为所有价格输入框添加事件监听
            document.querySelectorAll('.price-input').forEach(input => {
                input.addEventListener('input', handlePriceInput);
            });

            statsContainer.style.display = 'block';
            setTimeout(() => {
                statsContainer.scrollIntoView({ behavior: 'smooth', block: 'nearest' });
            }, 100);
        }

        function handlePriceInput(e) {
            const input = e.target;
            const name = input.dataset.name;
            const weight = parseFloat(input.dataset.weight);
            const price = parseFloat(input.value);
            
            if (price && price > 0) {
                priceData[name] = price;
                const amount = weight * price;
                const amountCell = document.querySelector(`.amount-cell[data-name="${name}"]`);
                if (amountCell) {
                    amountCell.textContent = `¥${amount.toFixed(2)}`;
                }
            } else {
                delete priceData[name];
                const amountCell = document.querySelector(`.amount-cell[data-name="${name}"]`);
                if (amountCell) {
                    amountCell.textContent = '--';
                }
            }
            
            updateTotalAmount();
        }

        function updateTotalInfo(grandTotal) {
            const totalAmount = calculateTotalAmount();
            const amountText = totalAmount !== null ? `，总金额 ¥${totalAmount.toFixed(2)}` : '';
            totalInfo.textContent = `共 ${currentStats.length} 个品名，总计 ${grandTotal.toFixed(2)} ${currentUnit}${amountText}`;
        }

        function calculateTotalAmount() {
            if (Object.keys(priceData).length === 0) return null;
            
            // 检查是否所有品名都有单价
            const allHavePrice = currentStats.every(item => priceData[item.name] !== undefined);
            if (!allHavePrice) return null;
            
            let total = 0;
            currentStats.forEach(item => {
                const price = priceData[item.name];
                if (price) {
                    total += item.totalWeight * price;
                }
            });
            
            return total;
        }

        function updateTotalAmount() {
            const grandTotal = currentStats.reduce((sum, item) => sum + item.totalWeight, 0);
            updateTotalInfo(grandTotal);
        }

        function copyResultsToClipboard() {
            const grandTotal = currentStats.reduce((sum, item) => sum + item.totalWeight, 0);

            // 从文件名提取时间
            let dateStr = '';
            const fileNameWithoutExt = currentFileName.replace(/\.(xlsx|xls)$/i, '');
            
            // 尝试匹配常见的日期格式：10-28、2024-10-28、20241028 等
            const datePatterns = [
                /(\d{1,2})-(\d{1,2})/,  // 10-28
                /(\d{4})-(\d{1,2})-(\d{1,2})/,  // 2024-10-28
                /(\d{4})(\d{2})(\d{2})/,  // 20241028
                /(\d{1,2})月(\d{1,2})/,  // 10月28
            ];
            
            for (const pattern of datePatterns) {
                const match = fileNameWithoutExt.match(pattern);
                if (match) {
                    if (match.length === 3) {
                        // 10-28 格式
                        dateStr = `${match[1]}-${match[2]}`;
                    } else if (match.length === 4) {
                        // 2024-10-28 格式
                        dateStr = `${match[1]}-${match[2]}-${match[3]}`;
                    }
                    break;
                }
            }

            const totalAmount = calculateTotalAmount();
            
            let textFormat = '';
            textFormat += '🥬 蔬菜统计结果';
            if (dateStr) {
                textFormat += ` (${dateStr})`;
            }
            textFormat += '\n';
            textFormat += '='.repeat(40) + '\n\n';
            textFormat += `共 ${currentStats.length} 个品名，总计 ${grandTotal.toFixed(2)} ${currentUnit}`;
            if (totalAmount !== null) {
                textFormat += `，总金额 ¥${totalAmount.toFixed(2)}`;
            }
            textFormat += '\n\n';

            currentStats.forEach((item, index) => {
                const price = priceData[item.name];
                const amount = price ? item.totalWeight * price : null;
                
                textFormat += `${(index + 1).toString().padStart(2, ' ')}. ${item.name.padEnd(15, ' ')} `;
                textFormat += `${item.totalWeight.toFixed(2)}${currentUnit}(${item.count}条)`;
                
                if (price) {
                    textFormat += ` @¥${price.toFixed(2)} = ¥${amount.toFixed(2)}`;
                }
                textFormat += '\n';
            });

            if (navigator.clipboard && navigator.clipboard.writeText) {
                navigator.clipboard.writeText(textFormat)
                    .then(() => {
                        showCopySuccess();
                    })
                    .catch(err => {
                        fallbackCopy(textFormat);
                    });
            } else {
                fallbackCopy(textFormat);
            }
        }

        function fallbackCopy(text) {
            const textarea = document.createElement('textarea');
            textarea.value = text;
            textarea.style.position = 'fixed';
            textarea.style.opacity = '0';
            document.body.appendChild(textarea);
            textarea.select();
            
            try {
                const successful = document.execCommand('copy');
                if (successful) {
                    showCopySuccess();
                } else {
                    showError('复制失败，请手动复制');
                }
            } catch (err) {
                showError('复制失败: ' + err.message);
            }
            
            document.body.removeChild(textarea);
        }

        function showCopySuccess() {
            copyBtn.classList.add('copied');
            copyIcon.textContent = '✓';
            copyText.textContent = '已复制';
            showSuccess('✅ 统计结果已复制到剪贴板！');

            setTimeout(() => {
                copyBtn.classList.remove('copied');
                copyIcon.textContent = '📋';
                copyText.textContent = '复制结果';
            }, 3000);
        }
    })();
</script>
